const app: Express = express();
const serveWeb = process.env.SERVE_WEB === "true";
const webDist = path.resolve(process.cwd(), "../ai-job-agent/dist/public");
app.set("trust proxy", 1);
app.disable("x-powered-by");
app.use((_req, res, next) => {
  res.setHeader("X-Content-Type-Options", "nosniff");
  res.setHeader("Referrer-Policy", "no-referrer");
  res.setHeader("X-Frame-Options", "DENY");
  res.setHeader("Permissions-Policy", "camera=(), microphone=(), geolocation=()");
  if (_req.path.startsWith("/api")) res.setHeader("Cache-Control", "no-store");
  res.setHeader(
    "Content-Security-Policy",
    serveWeb
      ? "default-src 'self'; script-src 'self'; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; font-src 'self' https://fonts.gstatic.com data:; img-src 'self' data: blob:; connect-src 'self'; object-src 'none'; base-uri 'self'; frame-ancestors 'none'"
      : "default-src 'none'; frame-ancestors 'none'; base-uri 'none'",
  );
  next();

app.use(
  pinoHttp({
    logger,
    serializers: {
      req(req) {
        return {
          id: req.id,
          method: req.method,
          url: req.url?.split("?")[0],
        };
      },
      res(res) {
        return {
          statusCode: res.statusCode,
        };
      },
    },
  }),
);

app.use(express.json({ limit: "1mb" }));
app.use(express.urlencoded({ extended: true }));

  app.use(express.static(webDist, { index: false }));

  app.get("/{*splat}", (req, res, next) => {
    if (req.path === "/api" || req.path.startsWith("/api/")) {
      next();
      return;
    }
    res.sendFile(path.join(webDist, "index.html"), (error) => {
      if (error) next(error);
    });

  req.log.error({ err: error }, "Unhandled API error");

    typeof error === "object" && error !== null && "code" in error
      ? String(error.code)
      : "";

    status === 413

  ControlAgentBody,
  ControlAgentResponse,
  GetAgentStatusResponse,
} from "@workspace/api-zod";
import { db, systemStateTable } from "@workspace/db";

const router: IRouter = Router();

async function getState() {
  const [state] = await db
    .select()
    .from(systemStateTable)
    .where(eq(systemStateTable.id, "agent"))
    .limit(1);
  return state;
}

router.get("/agent/status", async (_req, res): Promise<void> => {
  const state = await getState();
  res.json(
    GetAgentStatusResponse.parse({
      state: state?.agentState ?? "STOPPED",
      message: state?.agentMessage ?? "Background workers are not configured yet.",
      updatedAt: (state?.updatedAt ?? new Date()).toISOString(),
    }),
  );

router.post("/agent/control", async (req, res): Promise<void> => {
  const parsed = ControlAgentBody.safeParse(req.body);
  if (!parsed.success) {
    res.status(400).json({ error: parsed.error.message });
    return;
  }
  if (parsed.data.command !== "stop") {
    res.status(409).json({
      error: "Discovery workers and authorized job sources are not configured yet. The agent remains stopped.",
    });
    return;
  }

  const updatedAt = new Date();
  const [state] = await db
    .insert(systemStateTable)
    .values({
      id: "agent",
      agentState: "STOPPED",
      agentMessage: "Stopped. No automated applications or submissions can run.",
      updatedAt,
    })
    .onConflictDoUpdate({
      target: systemStateTable.id,
      set: {
        agentState: "STOPPED",
        agentMessage: "Stopped. No automated applications or submissions can run.",
        updatedAt,
      },
    })
    .returning();
  res.json(
    ControlAgentResponse.parse({
      state: state.agentState,
      message: state.agentMessage,
      updatedAt: state.updatedAt.toISOString(),
    }),
  );
});

import { Router, type IRouter } from "express";
import { and, desc, eq, like, or } from "drizzle-orm";
import {
  CreateJobBody,
  CreateJobResponse,
  GetJobParams,
  GetJobResponse,
  ListJobsQueryParams,
  ListJobsResponse,
  TrackJobApplicationBody,
  TrackJobApplicationParams,
  TrackJobApplicationResponse,
  UpdateJobBody,
  UpdateJobParams,
  UpdateJobResponse,
} from "@workspace/api-zod";
import { activityTable, applicationsTable, db, jobsTable } from "@workspace/db";
import { fingerprintJob, isPotentialDuplicate, normalizeCompany, normalizeText } from "../lib/job-deduplication";

const router: IRouter = Router();

function uniqueConflict(error: unknown): boolean {
  return (
    typeof error === "object" &&
    error !== null &&
    "code" in error &&
    typeof error.code === "string" &&
    error.code.startsWith("SQLITE_CONSTRAINT")
  );
}

function duplicateResponse(res: Parameters<Parameters<typeof router.post>[1]>[1], jobId: string): void {
  res.status(409).json({
    error: "This looks like a job already in your list. No second record was created.",
    existingJobId: jobId,
  });
}

router.get("/jobs", async (req, res): Promise<void> => {
  const parsed = ListJobsQueryParams.safeParse(req.query);
  if (!parsed.success) {
    res.status(400).json({ error: parsed.error.message });
    return;
  }
  const filters = [];
  if (parsed.data.status) filters.push(eq(jobsTable.status, parsed.data.status));
  if (parsed.data.q?.trim()) {
    const term = `%${parsed.data.q.trim()}%`;
    filters.push(
      or(
        like(jobsTable.title, term),
        like(jobsTable.company, term),
        like(jobsTable.location, term),
        like(jobsTable.description, term),
      ),
    );
  }
  const rows = await db
    .select()
    .from(jobsTable)
    .where(filters.length ? and(...filters) : undefined)
    .orderBy(desc(jobsTable.discoveredAt))
    .limit(500);
  res.json(ListJobsResponse.parse(rows));
});

router.post("/jobs", async (req, res): Promise<void> => {
  const parsed = CreateJobBody.safeParse(req.body);
  if (!parsed.success) {
    res.status(400).json({ error: parsed.error.message });
    return;
  }

  const input = parsed.data;
  const normalizedCompany = normalizeCompany(input.company);
  const normalizedTitle = normalizeText(input.title);
  const normalizedLocation = normalizeText(input.location ?? "");
  const fingerprint = fingerprintJob(input);

  const [fingerprintMatch] = await db
    .select({ id: jobsTable.id })
    .from(jobsTable)
    .where(eq(jobsTable.fingerprint, fingerprint))
    .limit(1);
  if (fingerprintMatch) {
    duplicateResponse(res, fingerprintMatch.id);
    return;
  }
  if (input.externalJobId) {
    const [sourceMatch] = await db
      .select({ id: jobsTable.id })
      .from(jobsTable)
      .where(
        and(
          eq(jobsTable.source, input.source),
          eq(jobsTable.externalJobId, input.externalJobId),
        ),
      )
      .limit(1);
    if (sourceMatch) {
      duplicateResponse(res, sourceMatch.id);
      return;
    }
  }

  const id = crypto.randomUUID();
  const now = new Date();
  try {
    const [job] = await db
      .insert(jobsTable)
      .values({
        id,
        ...input,
        normalizedCompany,
        normalizedTitle,
        normalizedLocation,
        fingerprint,
        status: "DISCOVERED",
        discoveredAt: now,
      })
      .returning();
    await db.insert(activityTable).values({
      id: crypto.randomUUID(),
      kind: "job",
      message: `Added ${job.title} at ${job.company}.`,
      createdAt: now,
    });
    res.status(201).json(CreateJobResponse.parse(job));
  } catch (error) {
    if (uniqueConflict(error)) {
      const [existing] = await db
        .select({ id: jobsTable.id })
        .from(jobsTable)
        .where(
          input.externalJobId
            ? or(
                eq(jobsTable.fingerprint, fingerprint),
                and(
                  eq(jobsTable.source, input.source),
                  eq(jobsTable.externalJobId, input.externalJobId),
                ),
              )
            : eq(jobsTable.fingerprint, fingerprint),
        )
        .limit(1);
      if (existing) {
        duplicateResponse(res, existing.id);
        return;
      }
    }
    throw error;
  }
});

router.get("/jobs/:id", async (req, res): Promise<void> => {
  const params = GetJobParams.safeParse(req.params);
  if (!params.success) {
    res.status(400).json({ error: params.error.message });
    return;
  }
  const [job] = await db.select().from(jobsTable).where(eq(jobsTable.id, params.data.id));
  if (!job) {
    res.status(404).json({ error: "Job not found." });
    return;
  }
  res.json(GetJobResponse.parse(job));
});

router.patch("/jobs/:id", async (req, res): Promise<void> => {
  const params = UpdateJobParams.safeParse(req.params);
  const body = UpdateJobBody.safeParse(req.body);
  if (!params.success || !body.success) {
    res.status(400).json({ error: "Invalid job update." });
    return;
  }
  const [job] = await db
    .update(jobsTable)
    .set({ status: body.data.status })
    .where(eq(jobsTable.id, params.data.id))
    .returning();
  if (!job) {
    res.status(404).json({ error: "Job not found." });
    return;
  }
  await db.insert(activityTable).values({
    id: crypto.randomUUID(),
    kind: "job",
    message: `${job.title} at ${job.company} marked ${job.status.toLowerCase().replaceAll("_", " ")}.`,
    createdAt: new Date(),
  });
  res.json(UpdateJobResponse.parse(job));
});

router.post("/jobs/:id/track", async (req, res): Promise<void> => {
  const params = TrackJobApplicationParams.safeParse(req.params);
  const body = TrackJobApplicationBody.safeParse(req.body);
  if (!params.success || !body.success) {
    res.status(400).json({ error: "Invalid application tracking request." });
    return;
  }
  const [job] = await db.select().from(jobsTable).where(eq(jobsTable.id, params.data.id));
  if (!job) {
    res.status(404).json({ error: "Job not found." });
    return;
  }

  const [existingApplication] = await db
    .select({ id: applicationsTable.id })
    .from(applicationsTable)
    .where(eq(applicationsTable.jobId, job.id))
    .limit(1);
  if (existingApplication) {
    res.status(409).json({ error: "An application record already exists for this job." });
    return;
  }

  const existingApplications = await db
    .select({ job: jobsTable, application: applicationsTable })
    .from(applicationsTable)
    .innerJoin(jobsTable, eq(applicationsTable.jobId, jobsTable.id));
  const similarApplication = existingApplications.find(({ job: otherJob }) =>
    isPotentialDuplicate(job, otherJob),
  );
  if (similarApplication) {
    res.status(409).json({
      error: `A similar job already has an application record (${similarApplication.job.title} at ${similarApplication.job.company}). Review it before creating another.`,
    });
    return;
  }

  const now = new Date();
  const id = crypto.randomUUID();
  let application;
  try {
    [application] = await db
      .insert(applicationsTable)
      .values({
        id,
        jobId: job.id,
        method: body.data.method,
        status: "NEEDS_REVIEW",
        matchScore: job.matchScore,
        appliedAt: now,
        lastActivityAt: now,
      })
      .returning();
  } catch (error) {
    if (uniqueConflict(error)) {
      res.status(409).json({ error: "An application record already exists for this job." });
      return;
    }
    throw error;
  }
  await db.insert(activityTable).values({
    id: crypto.randomUUID(),
    kind: "application",
    message: `Prepared a review record for ${job.title} at ${job.company}. Nothing was submitted.`,
    createdAt: now,
  });
  res.status(201).json(
    TrackJobApplicationResponse.parse({
      ...application,
      jobTitle: job.title,
      company: job.company,
    }),
  );
});

export default router;

--- artifacts/api-server/src/lib/job-deduplication.ts
import { createHash } from "node:crypto";
import type { JobRecord } from "@workspace/db";

export function normalizeText(value: string): string {
  return value
    .normalize("NFKD")
    .replace(/\p{Diacritic}/gu, "")
    .toLowerCase()
    .replace(/\blimited\b|\bltd\b|\bllc\b|\bplc\b|\bgmbh\b/g, " ")
    .replace(/\bincorporated\b|\binc\b|\bcorporation\b|\bcorp\b/g, " ")
    .replace(/\bcompany\b|\bco\b|\bholdings\b|\bgroup\b/g, " ")
    .replace(/[^a-z0-9]+/g, " ")
    .trim()
    .replace(/\s+/g, " ");
}

export function normalizeCompany(value: string): string {
  return normalizeText(value);
}

export function fingerprintJob(input: {
  company: string;
  title: string;
  location: string | null;
  description: string;
}): string {
  const canonical = JSON.stringify([
    normalizeCompany(input.company),
    normalizeText(input.title),
    normalizeText(input.location ?? ""),
    normalizeText(input.description),
  ]);
  return createHash("sha256").update(canonical).digest("hex");
}

function tokenSimilarity(left: string, right: string): number {
  const a = new Set(normalizeText(left).split(" ").filter((word) => word.length > 1));
  const b = new Set(normalizeText(right).split(" ").filter((word) => word.length > 1));
  if (a.size === 0 || b.size === 0) return 0;
  let overlap = 0;
  for (const word of a) {
    if (b.has(word)) overlap += 1;
  }
  return overlap / new Set([...a, ...b]).size;
}

export function isPotentialDuplicate(left: JobRecord, right: JobRecord): boolean {
  if (left.id === right.id || left.normalizedCompany !== right.normalizedCompany) {
    return false;
  }
  const sameLocation =
    left.normalizedLocation === right.normalizedLocation ||
    tokenSimilarity(left.location ?? "", right.location ?? "") >= 0.65;
  if (!sameLocation) return false;

  const titleSimilarity = tokenSimilarity(left.title, right.title);
  const descriptionSimilarity = tokenSimilarity(left.description, right.description);
  return (
    (titleSimilarity >= 0.75 && descriptionSimilarity >= 0.55) ||
    (titleSimilarity >= 0.4 && descriptionSimilarity >= 0.85)
  );
}

import mammoth from "mammoth";
import { getDocument } from "pdfjs-dist/legacy/build/pdf.mjs";

const maxPages = 60;
export const maxCvTextCharacters = 60_000;

export async function extractCvText(
  bytes: Buffer,
  fileType: "pdf" | "docx",
): Promise<{ text: string; truncated: boolean }> {
  let extracted: string;

  if (fileType === "docx") {
    const result = await mammoth.extractRawText({ buffer: bytes });
    extracted = result.value;
  } else {
    const document = await getDocument({
      data: new Uint8Array(bytes),
      isEvalSupported: false,
      useSystemFonts: true,
    }).promise;
    try {
      const pageCount = Math.min(document.numPages, maxPages);
      const pages: string[] = [];
      for (let pageNumber = 1; pageNumber <= pageCount; pageNumber += 1) {
        const page = await document.getPage(pageNumber);
        const content = await page.getTextContent();
        pages.push(
          content.items
            .map((item) => ("str" in item && typeof item.str === "string" ? item.str : ""))
            .filter(Boolean)
            .join(" "),
        );
      }
      extracted = pages.join("\n");
    } finally {
      await document.destroy();
    }
  }

  const normalized = extracted.replace(/\s+/g, " ").trim();
  if (!normalized) {
    return { text: "", truncated: false };
  }
  return {
    text: normalized.slice(0, maxCvTextCharacters),
    truncated: normalized.length > maxCvTextCharacters,
  };
}
